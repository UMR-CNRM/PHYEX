12def QSAT_SEAWATER(PT,PP,CQSAT):
13 cst = Cst()
14 ZFOES = PSAT(PT,CQSAT)
15 ZFOES = 0.98*ZFOES
16 # vapor pressure reduction of 2% over saline seawater could have a significant
17 # impact on the computation of surface latent heat flux under strong wind
18 # conditions (Zeng et al, 1998).
19 #
20 ZWORK1 = ZFOES/PP
21 ZWORK2 = cst.Rd/cst.Rv
22 PQSAT = ZWORK2*ZWORK1 / (1.+(ZWORK2-1.)*ZWORK1)
23 return PQSAT

33def PSAT(PT,CQSAT):
34 cst = Cst()
35 ZALP = cst.alpw
36 ZBETA = cst.betaw
37 ZGAM = cst.gamw
38 if(CQSAT=='NEW' and PT<=cst.Tt):
39 ZALP = cst.alpi
40 ZBETA = cst.betai
41 ZGAM = cst.gami
42
43 PPSAT = math.exp( ZALP - ZBETA/PT - ZGAM*math.log(PT) )
44 return PPSAT

46def CHARNOCK_WAGE(PWIND,PWAGE):
47 ZCOEFU = [ 0.70, -2.52 ]
48 ZCOEFA2 = [ 2.27, -6.67E-02 ]
49 ZCOEFB2 = [ -2.41, 4.30E-02 ]
50 ZCOEFA = [ -9.202, 2.265, -0.134, 2.35E-03 ]
51 ZCOEFB =[ -0.4124, -0.2225, 0.01178, -1.616E-04 ]
52 ZPOLYU = [ 0.0981, -4.13E-03, 4.34E-5, 1.16E-08 ]
53 ZLIMCHAR = 0.018
54 ZLIMCHAR2 = 0.002
55 ZLIMCHAR1 = 0.1
56 PCHARNWA = ZCOEFU[1-1]*(PWIND**ZCOEFU[2-1])
57 if PWIND >= 7.0:
58 ZAA = ZCOEFA[1-1] + ZCOEFA[2-1]*PWIND \
59 + ZCOEFA[3-1]*PWIND**2 \
60 + ZCOEFA[4-1]*PWIND**3
61 ZBB = ZCOEFB[1-1] + ZCOEFB[2-1]*PWIND \
62 + ZCOEFB[3-1]*PWIND**2 \
63 + ZCOEFB[4-1]*PWIND**3
64 PCHARNWA = ZAA * (PWAGE**ZBB)
65
66 if PWIND >= 23.0:
67 ZAA = ZCOEFA2[1-1] + ZCOEFA2[2-1]*PWIND
68 ZBB = ZCOEFB2[1-1] + ZCOEFB2[2-1]*PWIND
69 PCHARNWA= ZAA * (PWAGE**ZBB)
70 if PCHARNWA < ZLIMCHAR:
71 PCHARNWA = ZLIMCHAR
72
73 if PWIND >= 25.0:
74 PCHARNWA = ZPOLYU[1-1] + ZPOLYU[2-1]*PWIND \
75 + ZPOLYU[3-1]*PWIND**2 \
76 + ZPOLYU[4-1]*PWIND**3
77 if PCHARNWA < ZLIMCHAR2:
78 PCHARNWA = ZLIMCHAR2
79
80 # Final check, to avoid too large CHAR
81 if PCHARNWA > ZLIMCHAR1:
82 PCHARNWA = ZLIMCHAR1
83 return PCHARNWA

94def SURFACE_RI(XRIMAX, PTG, PQS, PEXNS, PEXNA, PTA, PQA,PZREF, PUREF,
95 PDIRCOSZW, PVMOD,XCISMIN, XVMODMIN, LALDTHRES ):
96 cst = Cst()
97 ZTHVA=PTA/PEXNA*( 1.+(cst.Rv/cst.Rd-1.)*PQA )
98 ZTHVS=PTG/PEXNS*( 1.+(cst.Rv/cst.Rd-1.)*PQS )
99 ZVMOD = WIND_THRESHOLD(PVMOD,PUREF,XCISMIN, XVMODMIN, LALDTHRES)
100 PRI = cst.g * PDIRCOSZW * PUREF * PUREF \
101 * (ZTHVA-ZTHVS) / (0.5 * (ZTHVA+ZTHVS) ) \
102 / (ZVMOD*ZVMOD) /PZREF
103 PRI = min(PRI,XRIMAX)
104 return PRI

107def PSifCTUW(PZL):
108 if PZL<0.:
109 ZX = math.pow(1.0 - 15. * PZL, 0.25) # Kansas unstable
110 ZPSIK= 2.0 * math.log((1.0+ZX )/2.0) \
111 + math.log((1.0+ZX*ZX)/2.0) \
112 - 2.0 * math.atan(ZX) \
113 + 2.0 * math.atan(1.0)
114 #
115 ZY= (1.0 - 10.15 * PZL)**0.3333 # Convective
116 ZPSIC= 1.5 * math.log((ZY*ZY+ZY+1.)/3.) \
117 - (3.0**0.5) * math.atan((2.0*ZY+1.0)/(3.0**0.5)) \
118 + 4.0 * math.atan(1.0)/(3.0**0.5)
119 #
120 ZF=PZL * PZL / (1.0+PZL*PZL)
121 #
122 PSIFCTUW=(1.-ZF) * ZPSIK + ZF * ZPSIC
123 else:
124 ZC=min(50.,0.35*PZL) # Stable
125 PSIFCTUW=-((1.+1.*PZL) + 0.6667*(PZL-14.28)/math.exp(ZC) + 8.525)
126 return PSIFCTUW

128def PSifCTTW(PZL):
129 if PZL<0.:
130 ZX = (1. - 15. * PZL)**.5 # Kansas unstable
131 ZPSIK= 2.0 * math.log((1.0+ZX )/2.0)
132 #
133 ZY = math.pow(1.0 - 34.15 * PZL,0.3333) # Convective
134 ZPSIC= 1.5 * math.log((ZY*ZY+ZY+1.0)/3.) \
135 - (3.0**0.5) * math.atan((2.0*ZY+1.0)/(math.pow(3.0,0.5))) \
136 + 4.0 * math.atan(1.0)/(3.0**0.5)
137 #
138 ZF = PZL * PZL / (1.0+PZL*PZL)
139 #
140 PSIFCTTW= (1.-ZF) * ZPSIK + ZF * ZPSIC
141 else:
142 ZC=min(50.,0.35*PZL) # Stable
143 PSIFCTTW=-(math.pow(1.+2.*PZL/3.,1.5) + 0.6667*(PZL-14.28)/math.exp(ZC) + 8.525)
144 return PSIFCTTW

146def WASP_FLUX(PTA,PQA, PEXNA, PRHOA, PVMOD, PZREF, PUREF, PSST, PEXNS, PPS, PRAIN, PHS, PTP,
147 XVZ0CM=0.,XRIMAX=0.2,XCISMIN=6.7E-5, XVMODMIN=0., LALDTHRES=False,CQSAT="NEW"):
148 cst = Cst()
149
150 ZRVSRDM1 = cst.Rv/cst.Rd-1. # 0.607766
151 ZRDSRV = cst.Rd/cst.Rv # 0.62198
152 ZBETAGUST = 1.2 # value based on TOGA-COARE experiment
153 ZZBL = 600. # Set a default value for boundary layer depth
154 ZS = 10.0 # Standard heigth =10m
155 ZCH10 = 0.00115
156 #
157 # 1.2 Array initialization by undefined values
158 #
159 PSFTH =0.0
160 PSFTQ =0.0
161 PUSTAR=0.0
162 #
163 PCD = 0.0
164 PCDN = 0.0
165 PCH = 0.0
166 PCE =0.0
167 PRI = 0.0
168 #
169 PRESA=0.0
170
171 XSURF_EPSILON=sys.float_info.epsilon
172 LPWG=False
173 LCPL_WAVE=False
174 LWAVEWIND=False
175 LPRECIP=False
176 #
177 #-------------------------------------------------------------------------------
178 # 2. INITIAL GUESS FOR THE ITERATIVE METHOD
179 # -------------------------------------
180 #
181 # 2.0 Temperature
182 #
183 # Set a non-zero value for the temperature gradient
184 #
185 if (PTA*PEXNS/PEXNA-PSST)==0.:
186 ZTA=PTA-1E-3
187 else:
188 ZTA=PTA
189
190 # 2.1 Wind and humidity
191 #
192 # Sea surface specific humidity
193 #
194 PQSAT=QSAT_SEAWATER(PSST,PPS,CQSAT)
195 #
196 # Set a minimum value to wind
197 #
198 ZVMOD = WIND_THRESHOLD(PVMOD,PUREF,XCISMIN, XVMODMIN, LALDTHRES)
199 #
200 # Specific humidity at saturation at the atm. level
201 #
202 ZPA = cst.P0* (math.pow(PEXNA,(cst.Cpd/cst.Rd)))
203 ZQASAT = QSAT(ZTA,ZPA,CQSAT)
204 #
205 #
206 ZO = 0.0001
207 ZWG = 0.
208 if LPWG: ZWG = 0.5
209 #
210 ZCHARN = 0.011
211 #
212 # 2.2 initial guess
213 #
214 ZDU = ZVMOD #wind speed difference with surface current(=0) (m/s)
215 #initial guess for gustiness factor
216 ZDT = -(ZTA/PEXNA) + (PSST/PEXNS) #potential temperature difference
217 ZDQ = PQSAT-PQA #specific humidity difference
218 #
219 ZDUWG = math.sqrt(math.pow(ZDU,2)+math.pow(ZWG,2)) #wind speed difference including gustiness ZWG
220 #
221 # 2.3 initialization of neutral coefficients
222 #
223 ZU10 = ZDUWG*math.log(ZS/ZO)/(PUREF/ZO)
224 ZUSR = 0.035*ZU10
225 ZVISA = 1.326E-5*(1.+6.542E-3*(ZTA-cst.Tt)+8.301E-6*(ZTA-cst.Tt)**2-4.84E-9*(ZTA-cst.Tt)**3) #Andrea (1989) CRREL Rep. 89-11
226 #
227 ZO10 = ZCHARN*ZUSR*ZUSR/cst.g+0.11*ZVISA/ZUSR
228 ZCD = (cst.Karman/math.log(PUREF/ZO10))**2 #drag coefficient
229 ZCD10= math.pow((cst.Karman/math.log(ZS/ZO10)),2)
230 ZCT10= ZCH10/math.sqrt(ZCD10)
231 ZOT10= ZS/math.exp(cst.Karman/ZCT10)
232 #
233 #-------------------------------------------------------------------------------
234 # Grachev and Fairall (JAM, 1997)
235 ZCT = cst.Karman/math.log(PZREF/ZOT10) #temperature transfer coefficient
236 ZCC = cst.Karman*ZCT/ZCD #z/L vs Rib linear coef.
237 #
238 ZRIBCU = -PUREF/(ZZBL*0.004*ZBETAGUST**3) #saturation or plateau Rib
239 ZRIBU = -cst.g*PUREF*(ZDT+ZRVSRDM1*ZTA*ZDQ)/(ZTA*ZDUWG**2)
240 #
241 if ZRIBU<0.:
242 ZETU = ZCC*ZRIBU/(1.+ZRIBU/ZRIBCU) #Unstable G and F
243 else:
244 ZETU = ZCC*ZRIBU/(1.+27./9.*ZRIBU/ZCC)#Stable
245 #
246 ZL10 = PUREF/ZETU #MO length
247 #
248
249 #
250 # First guess M-O stability dependent scaling params. (u*,T*,q*) to estimate ZO and z/L (ZZL)
251 ZUSR = ZDUWG*cst.Karman/(math.log(PUREF/ZO10)-PSifCTUW(PUREF/ZL10))
252 ZTSR = -ZDT*cst.Karman/(math.log(PZREF/ZOT10)-PSifCTTW(PZREF/ZL10))
253 ZQSR = -ZDQ*cst.Karman/(math.log(PZREF/ZOT10)-PSifCTTW(PZREF/ZL10))
254 #
255 ZZL = 0.0
256 #
257 #
258 if ZETU>50.:
259 ITERMAX = 1
260 else:
261 ITERMAX = 3 #number of iterations
262 #
263 # 3. ITERATIVE LOOP TO COMPUTE USR, TSR, QSR
264 # -------------------------------------------
265 #
266 if (LWAVEWIND and not LCPL_WAVE):
267 ZTWAVE = 0.5*PVMOD
268 else:
269 ZTWAVE = PTP
270 # to avoid the nullity of HS and TP
271 if ZTWAVE == 0.0: ZTWAVE = 0.5*PVMOD
272 if ZTWAVE > 30.0: ZTWAVE = 0.5*PVMOD
273 ZCWAVE = cst.g*ZTWAVE/(2.*math.pi)
274 ZWAGE = ZCWAVE/ZUSR
275 #
276 #
277 #
278 # Boucle principale
279 for JLOOP in range(1, ITERMAX + 1):
280 if JLOOP > ITERMAX:
281 continue
282
283 ZCHARN = CHARNOCK_WAGE(ZVMOD, ZWAGE)
284
285 ZO = ZCHARN * math.pow(ZUSR,2) / cst.g + 0.11 * ZVISA / ZUSR # Smith 1988
286
287 ZOT = PZREF * math.exp(-(cst.Karman**2) / (ZCH10 * math.log(PUREF / ZO)))
288 ZOQ = ZOT
289
290 ZZL = cst.Karman * cst.g * PUREF * (ZTSR * (1. + ZRVSRDM1 * PQA) + ZRVSRDM1 * ZTA * ZQSR) / (ZTA * ZUSR**2 * (1. + ZRVSRDM1 * PQA))
291
292 ZZTL = ZZL * PZREF / PUREF # for T
293
294 ZPUZ = PSifCTUW(ZZL)
295 ZPTZ = PSifCTTW(ZZTL)
296 ZPQZ = ZPTZ # simplification
297
298 # 3.1 Scale parameters
299 ZUSR = ZDUWG * cst.Karman / (math.log(PUREF / ZO) - ZPUZ)
300 ZTSR = -ZDT * cst.Karman / (math.log(PZREF / ZOT) - ZPTZ)
301 ZQSR = -ZDQ * cst.Karman / (math.log(PZREF / ZOQ) - ZPQZ)
302
303 # 3.2 Gustiness factor (ZWG)
304 if LPWG:
305 ZBF = -cst.g / ZTA * ZUSR * (ZTSR + ZRVSRDM1 * ZTA * ZQSR)
306 if ZBF > 0.0:
307 ZWG = ZBETAGUST * (ZBF * ZZBL)**(1. / 3.)
308 else:
309 ZWG = 0.2
310 ZDUWG = math.sqrt(ZVMOD**2 + ZWG**2)
311
312 # Initialisation des flux
313 ZTAU = 0.0
314 ZHF = 0.0
315 ZEF = 0.0
316 ZTAUR = 0.0
317 ZRF = 0.0
318
319 # 4.1 Coefficients de transfert
320 PCD = (ZUSR / ZDUWG)**2
321 PCH = ZUSR * ZTSR / (ZDUWG * (ZTA * PEXNS / PEXNA - PSST))
322 PCE = ZUSR * ZQSR / (ZDUWG * (PQA - PQSAT))
323
324 PCDN = (cst.Karman / math.log(ZS / ZO))**2
325
326 PZ0SEA = ZCHARN * ZUSR**2 / cst.g + XVZ0CM * PCD / PCDN
327
328 ZLV = cst.LvTt + (cst.Cpv - cst.Cl) * (PSST - cst.Tt)
329
330 # 4.2 Flux de surface
331 ZTSR = -ZTSR
332 ZQSR = -ZQSR
333 ZTAU = -PRHOA * ZUSR**2 * ZVMOD / ZDUWG
334 ZHF = PRHOA * cst.Cpd * ZUSR * ZTSR
335 ZEF = PRHOA * ZLV * ZUSR * ZQSR
336
337 # 4.3 Flux de pluie
338 if LPRECIP:
339 ZTAC = ZTA - cst.Tt
340 ZXLR = cst.LvTt + (cst.Cpv - cst.Cl) * ZTAC
341 ZDQSDT = ZQASAT * ZXLR / (cst.Rd * ZTA**2)
342 ZDTMP = (1.0 + 3.309e-3 * ZTAC - 1.44e-6 * ZTAC**2) * 0.02411 / (PRHOA * cst.Cpd)
343 ZDWAT = 2.11e-5 * (cst.P0 / ZPA) * (ZTA / cst.Tt)**1.94
344
345 ZALFAC = 1.0 / (1.0 + ZRDSRV * ZDQSDT * ZXLR * ZDWAT / (ZDTMP * cst.Cpd))
346
347 ZCPLW = 4224.8482 + ZTAC * (-4.707 + ZTAC * (0.08499 + ZTAC * (1.2826e-3 + ZTAC * (4.7884e-5 - 2.0027e-6 * ZTAC))))
348
349 ZRF = PRAIN * ZCPLW * ZALFAC * (PSST - ZTA + (PQSAT - PQA) * ZXLR / cst.Cpd)
350 ZTAUR = -0.85 * (PRAIN * ZVMOD)
351
352
353 # 4.5 Friction velocity (avec pluie)
354 ZUSTAR2 = - (ZTAU + ZTAUR) / PRHOA
355 PUSTAR = math.sqrt(ZUSTAR2)
356
357 # 4.6 Flux totaux de surface
358 PSFTH = ZHF + ZRF
359 PSFTQ = ZEF / ZLV
360
361 # 5.1 Nombre de Richardson
362 ZDIRCOSZW = 1.0
363 PRI = SURFACE_RI(XRIMAX, PSST, PQSAT, PEXNS, PEXNA, ZTA, ZQASAT, PZREF, PUREF, ZDIRCOSZW, PVMOD,XCISMIN, XVMODMIN, LALDTHRES)
364
365 # 5.2 Conductance et résistance aérodynamiques
366 ZAC = PCH * ZVMOD
367 PRESA = 1. / max(ZAC, XSURF_EPSILON)
368
369 # 5.3 Hauteurs de rugosité sur mer
370 PZ0SEA = ZCHARN * ZUSTAR2 / cst.g + XVZ0CM * PCD / PCDN
371 PZ0HSEA = PZ0SEA
372
373 # PSFTH W m-2
374 # PSFTQ kg m-2 s-1
375 return PSFTH, PSFTQ, PUSTAR#, PQSAT, PCD, PCDN, PCH, PCE, PRI, PRESA, PZ0HSEA, PZ0SEA
