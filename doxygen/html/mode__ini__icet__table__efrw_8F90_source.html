13!! PURPOSE
14!! -------
15!! The purpose of this routine is to initialize the t_Efrw table for
16!! the ICE-T implementation.
17!!
18!!** METHOD
19!! ------
20!! Variable collision efficiency for rain collecting cloud water using
21!! method of Beard and Grover, 1974 if a/A less than 0.25; otherwise
22!! uses polynomials to get close match of Pruppacher & Klett Fig 14-9.
23!!
24!! EXTERNAL

26!! None

110 & *ice_t_parameters%XITDR(ji) &

112 & *ice_t_parameters%XITDR(ji) &
113 & *ice_t_parameters%XITDR(ji) &
114 & - 0.002362e12*ice_t_parameters%XITDR(ji) &
115 & *ice_t_parameters%XITDR(ji) &
116 & *ice_t_parameters%XITDR(ji) &
117 & *ice_t_parameters%XITDR(ji)
118
119 zstokes = ice_t_parameters%XITDC(jj)*ice_t_parameters%XITDC(jj)*zvtr*xrholw/(9.*1.718e-5*ice_t_parameters%XITDR(ji))
120 zreynolds = 9.*zstokes/(zp*zp*xrholw)
121
122 zf = log(zreynolds)
123 zg = -0.1007d0 - 0.358d0*zf + 0.0261d0*zf*zf
124 zk0 = exp(zg)
125 zz = log(zstokes/(zk0+1.d-15))
126 zh = 0.1465d0 + 1.302d0*zz - 0.607d0*zz*zz + 0.293d0*zz*zz*zz
127 zyc0 = 2.0d0/xpi * atan(zh)
128 zef_rw = (zyc0+zp)*(zyc0+zp) / ((1.+zp)*(1.+zp))
129
130 ENDIF
131
132 ice_t_parameters%XT_EFRW(ji,jj) = max(0.0, min(real(zef_rw), 0.95))
