19class Grid():
20 """
21 The Grid class handles vertical grids
22
23 The definition of the vertical grids are stored in files with the '.grid' extension.
24 Theses files contains a dictionary as a json string with the folowing content:
25 - kind: 'hybridP' for an hybrid-pressure coordinate (pressure on level i is P(i)=a(i)+b(i)*Ps)
26 'hybridH' for an hybrid-height coordinate (altitude on level i is Z(i)=a(i)+b(i)*Zs)
27 'P' for a pressure coordinate
28 'H' for a height (above sea level) coordinate
29 - position: 'FLUX' if the grid refers to flux levels
30 'MASS' otherwise
31 - mean: 'linear' or 'quadratic'
32 way to compute flux levels from mass levels if grid is defined on mass levels
33 or the reverse if grid is defined on flux levels
34 - description: a single list of values if kind is 'P' or 'H'
35 a list containing a list for the a(i) coefficients and a list for the b(i)
36 coefficients if kind is 'hybridP' or 'hybridH'.
37 """

98 def _mean(self, values, method):
99 """
100 Computes the mean
101 :param method: must be 'linear' or 'quadartic'
102 """
103 if method == 'linear':
104 values = numpy.array([.5 * (values[i] + values[i + 1])
105 for i in range(self.len - 1)])
106 elif method == 'quadratic':
107 if values[0] == 0.:
108 update = (0, values[1] / 2.) # Linear when quadratic is not possible
109 elif values[-1] == 0.:
110 update = (-1, values[-2] / 2.) # Linear when quadratic is not possible
111 else:
112 update = None
113 values = numpy.array([numpy.sqrt(values[i] * values[i + 1])
114 for i in range(self.len - 1)])
115 if update is not None:
116 values[update[0]] = update[1]
117 else:
118 raise Phyex1DError("method must be 'linear' or 'quadartic'")
119 return values

121 def get_pressure(self, position, state, pronostic_variables):
122 """
123 Returns the level pressures
124 :param position: must be among grid.MASS + .grid.FLUX
125 :param state: dictionary containing the current state
126 :param pronostic_variables: list of pronostic variables
127 """
128 if self.kind == 'hybridP' and position == self.position:
129 if 'Ps' not in state:
130 raise Phyex1DError("Surface pressure ('Ps') is needed to compute the pressures")
131 pressure = numpy.array([self.description[0][i] + self.description[1][i] * state['Ps']
132 for i in range(self.len)])
133 if self.ascending is None:
134 self.ascending = numpy.all(pressure[1:] < pressure[:-1])
135 elif self.kind == 'P' and position == self.position:
136 pressure = numpy.array(self.description)
137 if self.ascending is None:
138 self.ascending = numpy.all(pressure[1:] < pressure[:-1])
139 elif self.kind in ('P', 'hybridP'):
140 pressure = self._mean(self.get_pressure(self.position, state, pronostic_variables),
141 self.mean)
142 elif self.kind in ('H', 'hybridH'):
143 # We assume hydrostatic equilibrium
144 # leading to dln(P) / dz = -g / (R * T) if T is the prognostic variable
145 # or d[(P / P0) ** (Rd / Cpd)] / dz = -g / (R * Theta) * (Rd / Cpd) with Theta
146 if 'qv' in pronostic_variables:
147 gas_constant = self.cst.Rd + state['qv'] * (self.cst.Rv - self.cst.Rd)
148 for var in ('qc', 'qr', 'qi', 'qs', 'qg', 'qh'):
149 if var in pronostic_variables:
150 gas_constant += - state[var] * self.cst.Rd
151 elif 'rv' in pronostic_variables:
152 div = 1 + state['rv']
153 for var in ('rc', 'rr', 'ri', 'rs', 'rg', 'rh'):
154 if var in pronostic_variables:
155 div += state[var]
156 gas_constant = (self.cst.Rd + state['rv'] * self.cst.Rv) / div
157 z_flux = self.get_altitude('FLUX', state, pronostic_variables)
158 dx = - self.cst.g / gas_constant * numpy.diff(z_flux)
159 if 'T' in pronostic_variables:
160 dx = dx / state['T']
161 elif 'Theta' in pronostic_variables:
162 dx = dx / state['Theta'] * self.cst.Rd / self.cst.Cpd
163 else:
164 raise Phyex1DError("T or Theta must be prognostic")
165 x_flux = numpy.zeros(z_flux.shape)
166 if self.ascending:
167 x_flux[1:] = numpy.cumsum(dx)
168 else:
169 x_flux[:-1] = numpy.cumsum(-dx[::-1])[::-1]
170 if 'T' in pronostic_variables:
171 x_flux += numpy.log(state['Ps'])
172 pressure_flux = numpy.exp(x_flux)
173 else:
174 x_flux += (state['Ps'] / 1.E5) ** (self.cst.Rd / self.cst.Cpd)
175 pressure_flux = x_flux ** (self.cst.Cpd / self.cst.Rd) * 1.E5
176 if position == 'FLUX':
177 pressure = pressure_flux
178 else:
179 # We add the contribution between flux and mass levels assuming constant profiles
180 z_mass = self.get_altitude('MASS', state, pronostic_variables)
181 dx = - self.cst.g / gas_constant
182 if self.ascending:
183 dx = dx * (z_mass - z_flux[:-1])
184 else:
185 dx = dx * (z_mass - z_flux[1:])
186 if self.ascending:
187 temp = pressure_flux[:-1]
188 else:
189 temp = pressure_flux[1:]
190 if 'T' in pronostic_variables:
191 dx = dx / state['T']
192 pressure = numpy.exp(numpy.log(temp) + dx)
193 elif 'Theta' in pronostic_variables:
194 dx = dx / state['Theta'] * self.cst.Rd / self.cst.Cpd
195 pressure = ((temp / 1.E5) ** (self.cst.Rd / self.cst.Cpd) + dx
196 ) ** (self.cst.Cpd / self.cst.Rd) * 1.E5
197 else:
198 raise Phyex1DError("T or Theta must be prognostic")
199 else:
200 raise Phyex1DError("Wrong grid's kind")
201
202 return pressure

204 def get_altitude(self, position, state, pronostic_variables):
205 """
206 Returns the level altitudes
207 :param position: must be among grid.MASS + .grid.FLUX
208 :param state: dictionary containing the current state
209 :param pronostic_variables: list of pronostic variables
210 """
211 if self.kind == 'hybridH' and position == self.position:
212 if 'Zs' not in state:
213 raise Phyex1DError("Surface altitude ('Zs') is needed to compute the altitudes")
214 altitude = numpy.array([self.description[0][i] + self.description[1][i] * state['Zs']
215 for i in range(self.len)])
216 if self.ascending is None:
217 self.ascending = numpy.all(altitude[1:] > altitude[:-1])
218 elif self.kind == 'H' and position == self.position:
219 altitude = numpy.array(self.description)
220 if self.ascending is None:
221 self.ascending = numpy.all(altitude[1:] > altitude[:-1])
222 elif self.kind in ('H', 'hybridH'):
223 altitude = self._mean(self.get_altitude(self.position, state, pronostic_variables),
224 self.mean)
225 elif self.kind in ('P', 'hybridP'):
226 # We assume hydrostatic equilibrium
227 # leading to dln(P) / dz = -g / (R*T)
228 pressure_mass = None
229 if 'T' in pronostic_variables:
230 temperature = state['T']
231 elif 'Theta' in pronostic_variables:
232 pressure_mass = self.get_pressure('MASS', state, pronostic_variables)
233 exner = (pressure_mass / 1.E5) ** (self.cst.Rd / self.cst.Cpd)
234 temperature = exner * state['Theta']
235 else:
236 raise Phyex1DError("T or Theta must be prognostic")
237 if 'qv' in pronostic_variables:
238 gas_constant = self.cst.Rd + state['qv'] * (self.cst.Rv - self.cst.Rd)
239 for var in ('qc', 'qr', 'qi', 'qs', 'qg', 'qh'):
240 if var in pronostic_variables:
241 gas_constant += - state[var] * self.cst.Rd
242 elif 'rv' in pronostic_variables:
243 div = 1 + state['rv']
244 for var in ('rc', 'rr', 'ri', 'rs', 'rg', 'rh'):
245 if var in pronostic_variables:
246 div += state[var]
247 gas_constant = (self.cst.Rd + state['rv'] * self.cst.Rv) / div
248 pressure_flux = self.get_pressure('FLUX', state, pronostic_variables)
249 dz = -gas_constant * temperature / self.cst.g * numpy.diff(numpy.log(pressure_flux))
250 z_flux = numpy.zeros(pressure_flux.shape)
251 if self.ascending:
252 z_flux[1:] = numpy.cumsum(dz)
253 else:
254 z_flux[:-1] = numpy.cumsum(-dz[::-1])[::-1]
255 z_flux += state['Zs']
256 if position == 'FLUX':
257 altitude = z_flux
258 else:
259 # We add the contribution between flux and mass levels assuming constant profiles
260 if pressure_mass is None:
261 pressure_mass = self.get_pressure('MASS', state, pronostic_variables)
262 if self.ascending:
263 dlnp = numpy.log(pressure_mass) - numpy.log(pressure_flux[:-1])
264 else:
265 dlnp = numpy.log(pressure_mass) - numpy.log(pressure_flux[1:])
266 dz = -gas_constant * temperature / self.cst.g * dlnp
267 if self.ascending:
268 altitude = z_flux[:-1] + dz
269 else:
270 altitude = z_flux[1:] + dz
271 else:
272 raise Phyex1DError("Wrong grid's kind")
273 return altitude
