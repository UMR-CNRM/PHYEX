25def make_idealised_dataset():
26 """
27 Create an xarray Dataset with idealised atmospheric profiles
28 on a 90-level height grid up to 20 km.
29 """
30 nz = 90
31 z = np.linspace(0, 20000, nz) # height above ground (m)
32 time = np.array([0., 3600.]) # start and end times (s)
33
34 # Lapse-rate temperature profile
35 T = np.broadcast_to(290. - z * 0.0065, (2, nz))
36 # Exponentially decreasing moisture
37 qv = np.broadcast_to(0.01 * np.exp(-z / 3000.), (2, nz))
38 # Zero wind
39 u = np.zeros((2, nz))
40 v = np.zeros((2, nz))
41 w = np.zeros((2, nz))
42 tke = np.zeros((2, nz))
43
44 ds = xr.Dataset(
45 {
46 'Zs': ('time', np.array([0., 0.])),
47 'zh': (('time', 'z'), np.broadcast_to(z, (2, nz))),
48 'orog': ('time', np.array([0., 0.])),
49 'Ps': ('time', np.array([101325., 101325.])),
50 'Ts': ('time', np.array([290., 290.])),
51 'T': (('time', 'z'), T),
52 'qv': (('time', 'z'), qv),
53 'u': (('time', 'z'), u),
54 'v': (('time', 'z'), v),
55 'w': (('time', 'z'), w),
56 'tke': (('time', 'z'), tke),
57 'lat': ('time', np.array([0., 0.])),
58 'lon': ('time', np.array([0., 0.])),
59 'ts_forc': ('time', np.array([290., 290.])),
60 'ps_forc': ('time', np.array([101325., 101325.])),
61 'zh_forc': (('time', 'z'), np.broadcast_to(z, (2, nz))),
62 },
63 coords={'time': time, 'z': z},
64 attrs={
65 'adv_theta': 0,
66 'adv_ta': 0,
67 'adv_rv': 0,
68 'adv_qv': 0,
69 'adv_ua': 0,
70 'adv_va': 0,
71 'nudging_theta': 0,
72 'nudging_ta': 0,
73 'nudging_rv': 0,
74 'nudging_qv': 0,
75 'nudging_ua': 0,
76 'nudging_va': 0,
77 'forc_wa': 0,
78 'forc_geo': 0,
79 'radiation': 'none',
80 'surface_forcing_temp': 'ts',
81 'surface_forcing_moisture': 'none',
82 'surface_forcing_wind': 'none',
83 'surface_type': 'ocean',
84 'start_date': '2000-01-01T00:00:00',
85 },
86 )
87 return ds

90def make_plots(runner, output_dir):
91 """Create plots similar to the easy_install script."""
92 plots = [
93 ('T.png', ['T'], {'y_var': 'P'}),
94 ('qv.png', ['qv'], {'y_var': 'P'}),
95 ('qc.png', ['qc'], {'y_var': 'P', 'vmin': 0., 'vmax': 5.e-5}),
96 ('Theta.png', ['Theta'], {'y_var': 'P'}),
97 ]
98 for filename, var_names, options in plots:
99 fig, ax = plt.subplots()
100 kwargs_contourf = {}
101 vmin = options.pop('vmin', None)
102 vmax = options.pop('vmax', None)
103 if vmin is not None:
104 kwargs_contourf['vmin'] = vmin
105 if vmax is not None:
106 kwargs_contourf['vmax'] = vmax
107 if kwargs_contourf:
108 options['kwargs_contourf'] = kwargs_contourf
109 runner.plot_evol(ax, var_names, **options)
110 if options.get('y_var', 'X') == 'P':
111 ax.invert_yaxis()
112 filepath = os.path.join(output_dir, filename)
113 fig.savefig(filepath)
114 plt.close(fig)
115 print(f' Saved {filepath}')
