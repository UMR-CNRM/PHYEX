6ELEMENTAL SUBROUTINE criauti(PRCRIAUTI, PT0CRIAUTI, PACRIAUTI, PBCRIAUTI)
7!!
8!!** PURPOSE
9!! -------
10!! Compute the A and B coefficients of the 10**(A*T+B) law used for the
11!! ice->snow autoconversion threshold (cirrus law), from the threshold
12!! value PRCRIAUTI (possibly perturbed, e.g. by SPP) and the temperature
13!! PT0CRIAUTI at which the cirrus law starts to be used.
14!! The second point needed to determine the law is fixed (-40 degC, 1.25E-6).
15!!
16!! AUTHOR
17!! ------
18!! U. Andrae (2026)
19!!
20!! MODIFICATIONS
21!! -------------
22!!
23!
24!
25!* 0. DECLARATIONS
26! ------------
27!
28!$acc routine seq
29!
30IMPLICIT NONE
31!
32REAL, INTENT(IN) :: prcriauti ! Constants for pristine ice autoconversion
33REAL, INTENT(IN) :: pt0criauti ! Temp degC at which cirrus law starts to be used
34REAL, INTENT(OUT) :: pacriauti ! A Coef. for cirrus law
35REAL, INTENT(OUT) :: pbcriauti ! B Coef. for cirrus law
36!
37REAL :: ztcri0, zcri0
38!
39! Second point to determine 10**(aT+b) law
40!
41ztcri0=-40.0
42zcri0=1.25e-6
43!
44pbcriauti=-( log10(prcriauti) - log10(zcri0)*pt0criauti/ztcri0 )&
45 *ztcri0/(pt0criauti-ztcri0)
46pacriauti=(log10(zcri0)-pbcriauti)/ztcri0
47!
48END SUBROUTINE criauti
