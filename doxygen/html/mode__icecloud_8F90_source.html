17! Purpose:
18! calculate subgridscale fraction of supersaturation with respect to ice.
19! Method:
20! Assume a linear distubution of relative humidity and let the variability
21! of humidity be a function of model level thickness.
22! (Also a function of of humidity itself in the boundary layer)
23! Interface: subroutine ICECLOUD is called
24! ------------ from subroutine 'rain_ice'

33! PDZ : model level thickness (m)
34! PT : temperature (K)
35! PR : model level humidity mixing ratio (kg/kg)
36! PTSTEP : timestep
37! PPBLH : plantetary layer height (m) (negative value means unknown)
38! PWCLD : water and / mixed phase cloud cover (negative means unknown)
39! XW2D : quota between ice crystal concentration between dry and wet
40! part of a gridbox

42! OUTPUT arguments (arguments d'sortie)
43!---------------------------------------------
44! SIFRC : subgridscale fraction with supersaturation with respect to ice.
45! SSIO : Super-saturation with respect to ice in the
46! supersaturated fraction
47! SSIU : Sub-saturation with respect to ice in the sub-saturated

49! W2D : Factor used to get consistncy between the mean value of

162! input cloudcover:
163! IF(PWCLD < 0.99 .AND. PWCLD > 0.01) ZRHLIM= 1. - (1.-ZRHW)/(1.-PWCLD)**2

166 sifrc(jij,jk) = zwcld

169!ZRHLIMICE = MAX(ZRHDMIN*ZI2W,1.+ ZI2W*( ZRHLIM - 1.))
170 zrhlimice = 1.+ zi2w*( zrhlim - 1.)
171
172 IF(zrhlim <= 0.999)THEN
173
174 ! compute a 1/(1-rhlim) constistant with lstmp(i,k):
175 zrhliminv = 1./(1. - zrhlimice)
176 zrhdif = (zrhi - zrhlimice)*zrhliminv
177
178 IF(zwcld==0.)THEN
179 sifrc(jij,jk) = min(1.,0.5*max(0.,zrhdif))
180 ELSE
181 za = 1. - 1./zi2w
182 sifrc(jij,jk) = min(1.,za*0.5/ (1. - zrhlim))
183 sifrc(jij,jk) = min(1.,zwcld + sifrc(jij,jk))
184 ENDIF
185 ENDIF
186
187 IF(sifrc(jij,jk) > 0.01) THEN
188 ssiu(jij,jk) = sifrc(jij,jk) + zrhlimice*(1.-sifrc(jij,jk))
189 ssio(jij,jk) = (zrhi - (1.- sifrc(jij,jk))*ssiu(jij,jk))/sifrc(jij,jk)
190 ELSE
191 sifrc(jij,jk) = 0.! to aviod mismatch with output variables
192 za = min(0.,zrhi-zrhlimice)
193 ssiu(jij,jk) = max(0.,sifrc(jij,jk) + zrhlimice*(1.-sifrc(jij,jk)) + 2.*za )
194 ENDIF
195 ssio(jij,jk) = min(zi2w,ssio(jij,jk))
196 ssiu(jij,jk) = max(0.,ssiu(jij,jk))
197
198! Transform from relative humidity to degree of saturation:
199 ssiu(jij,jk) = ssiu(jij,jk) - 1.
200 ssio(jij,jk) = ssio(jij,jk) - 1.
201
202 IF (xw2d > 1.) w2d(jij,jk) = 1./(1. - sifrc(jij,jk) + xw2d*sifrc(jij,jk))
203 END IF
204 ENDDO
205ENDDO
