10!
11!!**** *INI_ICET_FREEZEH2O * - initialize the tpi_qcfz table for ICE-T
12!!
13!! PURPOSE
14!! -------
15!! The purpose of this routine is to initialize the tpi_qcfz table for
16!! the ICE-T implementation.
17!!
18!!** METHOD
19!! ------
20!! This is a literal adaptation of Bigg (1954) probability of drops of
21!! a particular volume freezing. Given this probability, simply freeze
22!! the proportion of drops summing their masses.
23!!
24!! EXTERNAL
25!! --------
26!! None
27!!
28!! IMPLICIT ARGUMENTS
29!! ------------------
30!! Module MODD_CST
31!! XRHOLW ! Liquid water density
32!! Module MODD_RAIN_ICE_DESCR
33!! XAR
34!! XBR

41!! XITDR
42!! XITDC
43!! XMU_R
44!! XN0R_EXP

46!! XTPI_QCFZ

48!! REFERENCE
49!! ---------
50!!
51!! AUTHOR
52!! ------
53!! B.J.K. Engdahl * MetNo *
54!!
55!! MODIFICATIONS
56!! -------------
57!! Original 2022

88 zprob, zvol, ztexp, zorho_w, &

99 zorho_w = 1./xrholw
100 DO jn2 = 1, nbr
101 zmassr(jn2) = rain_ice_descrn%XAR*ice_t_parameters%XITDR(jn2)**rain_ice_descrn%XBR
102 ENDDO
103 DO jn = 1, nbc
104 zmassc(jn) = rain_ice_descrn%XAR*ice_t_parameters%XITDC(jn)**rain_ice_descrn%XBR
105 ENDDO

107 !..Freeze water (smallest drops become cloud ice, otherwise graupel).
108 DO jm = 1, ntb_in
109 zt_adjust = max(-3.0, min(3.0 - log10(xnt_in(jm)), 3.0))
110 DO jk = 1, 45
111 ztexp = exp( dfloat(jk) - zt_adjust*1.0d0 ) - 1.0d0
112 DO jj = 1, ntb_r1
113 DO ji = 1, ntb_r
114 zlam_exp = (xn0r_exp(jj)*rain_ice_descrn%XAR*ice_t_parameters%XCR_GM(1)/xr_r(ji))**ice_t_parameters%XORE1
115 zlam_r = zlam_exp &
116 & * (ice_t_parameters%XCR_GM(3)*ice_t_parameters%XORG2*ice_t_parameters%XORG1)**ice_t_parameters%XOBMR
117 zn0_r = xn0r_exp(jj)/(ice_t_parameters%XCR_GM(2)*zlam_exp) * zlam_r**ice_t_parameters%XCR_EX(2)
118 zsum1 = 0.0d0
119 zsum2 = 0.0d0
120 zsumn1 = 0.0d0
121 zsumn2 = 0.0d0
122 DO jn2 = nbr, 1, -1
123 zn_r(jn2) = zn0_r*ice_t_parameters%XITDR(jn2)**xmu_r*exp(-zlam_r*ice_t_parameters%XITDR(jn2))&
124 & * ice_t_parameters%XITDTR(jn2)
125 zvol = zmassr(jn2)*zorho_w
126 zprob = 1.0d0 - exp(-120.0d0*zvol*5.2d-4 * ztexp)
127 IF (zmassr(jn2) .LT. ice_t_parameters%XM0G) THEN
128 zsumn1 = zsumn1 + zprob*zn_r(jn2)
129 zsum1 = zsum1 + zprob*zn_r(jn2)*zmassr(jn2)
130 ELSE
131 zsumn2 = zsumn2 + zprob*zn_r(jn2)
132 zsum2 = zsum2 + zprob*zn_r(jn2)*zmassr(jn2)
133 ENDIF
134 IF ((zsum1+zsum2) .GE. xr_r(ji)) EXIT
135 ENDDO
136 ice_t_parameters%XTPI_QRFZ(ji,jj,jk,jm) = zsum1
137 ice_t_parameters%XTPG_QRFZ(ji,jj,jk,jm) = zsum2
138 ENDDO
139 ENDDO
140
141 DO jj = 1, nbc
142 inu_c = min(15, nint(1000.e6/ice_t_parameters%XT_NC(jj)) + 2)
143 DO ji = 1, ntb_c
144 zlam_c = (ice_t_parameters%XT_NC(jj) * rain_ice_descrn%XAR * ice_t_parameters%XCC_GM(2,inu_c)&
145 & * ice_t_parameters%XOCG1(inu_c) / xr_c(ji))**ice_t_parameters%XOBMR
146 zn0_c = ice_t_parameters%XT_NC(jj) * ice_t_parameters%XOCG1(inu_c) * zlam_c**ice_t_parameters%XCC_EX(1,inu_c)
147 zsum1 = 0.0d0
148 zsumn2 = 0.0d0
149 DO jn = nbc, 1, -1
150 zvol = zmassc(jn)*zorho_w
151 zprob = 1.0d0 - exp(-120.0d0*zvol*5.2d-4 * ztexp)
152 zn_c(jn) = zn0_c*ice_t_parameters%XITDC(jn)**inu_c&
153 & * exp(-zlam_c*ice_t_parameters%XITDC(jn))*ice_t_parameters%XITDTC(jn)
154 zsum1 = zsum1 + zprob*zn_c(jn)*zmassc(jn)
155 IF (zsum1 .GE. xr_c(ji)) EXIT
156 ENDDO
157 ice_t_parameters%XTPI_QCFZ(ji,jj,jk,jm) = zsum1
158 ENDDO
159 ENDDO
160 ENDDO
161 ENDDO
162

165 IF (lhook) CALL dr_hook('INI_ICET_FREEZEH2O',1,zhook_handle)
