35def string2filename(string):
36 """
37 Tag (usable in file names) corresponding to a comparison or an experiment name:
38 quotes are removed, spaces and slashes are replaced by underscores
39 """

43def physics_options(cls):
44 """
45 Options of a physics class: the arguments of its constructor having a default
46 value, except the ones set by ExperimentRunner
47 :param cls: physics class
48 :return: dictionary {option name: default value}
49 """
50 return {name: param.default
51 for name, param in inspect.signature(cls.__init__).parameters.items()
52 if name not in RUNNER_ARGUMENTS and param.default is not inspect.Parameter.empty}

55class ExperimentRunner(PPPYComp):
56 """
57 Deals with the execution of the 1D model
58 """
59 def __init__(self, inputfile_or_case, experiments, output_dir, comp_name=DEFAULT_COMP_NAME):
60 """
61 :param inputfile_or_case: path to a netCDF driver file or a Case instance
62 :param experiments: list of dictionaries describing the experiments
63 allowed keys are:
64 - class: name of the class (key of PHYSICS_CLASSES,
65 DEFAULT_CLASS by default) or class implementing
66 the physics
67 - grid: Grid name or file name containing a grid description
68 (with '.grid' extension).
69 - dt: Timestep (s)
70 - name: experiment name
71 - attrs: dictionary of case attributes to override
72 the other keys are options given to the class (see
73 physics_options)
74 :param comp_name: comparison name (DEFAULT_COMP_NAME if empty)
75 The results are stored in <output_dir>/<comparison tag>/<experiment tag>.hdf5, the
76 tags being the comparison and experiment names with quotes removed and spaces and
77 slashes replaced by underscores.
78 """
79 if not comp_name:
80 comp_name = DEFAULT_COMP_NAME
81
82 if isinstance(inputfile_or_case, Case):
83 case = inputfile_or_case
84 else:
85 case = CaseCommonFormat(inputfile_or_case)
86
87 self.case = case
88 schemes = []
89 for exp in experiments:
90 cls = exp.get('class', DEFAULT_CLASS)
91 if isinstance(cls, str):
92 cls = PHYSICS_CLASSES[cls]
93 # The experiment keys are given to the class
94 kw = dict(exp)
95 kw.pop('class', None)
96 kw['dt'] = float(kw.get('dt', 1.)) # pppy needs a float
97 kw['method'] = 'step-by-step'
98 kw['name'] = kw.get('name', 'Experiment')
99 kw['tag'] = string2filename(kw['name'])
100 kw['case'] = case
101 kw['grid'] = Grid(kw.get('grid', 'L90arome'))
102 kw['attrs'] = kw.get('attrs', {})
103 schemes.append(cls(**kw))
104 super().__init__(schemes=schemes,
105 output_dir=output_dir,
106 duration=self.case.duration,
107 init_state={},
108 name=comp_name,
109 tag=string2filename(comp_name))

112def list_experiments(output_dir='.', comp_name=DEFAULT_COMP_NAME):
113 """
114 Existing results of a comparison
115 :param output_dir: output directory
116 :param comp_name: comparison name
117 :return: list of tuples (experiment name, physics class, result file); the name and
118 class are '?' for results produced by older versions
119 """
120 directory = os.path.join(output_dir, string2filename(comp_name or DEFAULT_COMP_NAME))
121 if not os.path.isdir(directory):
122 raise Phyex1DError(f"No results for the comparison {comp_name} ({directory} does "
123 "not exist)")
124 result = []
125 for filename in sorted(glob.glob(os.path.join(directory, '*.hdf5'))):
126 with h5py.File(filename, 'r') as hdf5:
127 result.append((str(hdf5.attrs.get('experiment_name', '?')),
128 str(hdf5.attrs.get('physics_class', '?')), filename))
129 return result
