486 def ustar2fluxes(ustar, u, v):
487 """
488 Convert ustar into momentum fluxes
489 :param ustar: ustar value
490 :param u, v: wind components
491 :return: w'u', w'v'
492 """
493 alpha = numpy.arctan2(v, u)
494 return - ustar**2 * numpy.cos(alpha), - ustar**2 * numpy.sin(alpha)
