217 def get_interpolator(self, varname, grid):
218 """
219 Get from cache or build an interpolator for a forcing variable
220
221 Parameters
222 ----------
223 varname : str
224 Variable name
225 grid : Grid
226 Grid instance for vertical coordinate
227
228 Returns
229 -------
230 RegularGridInterpolator
231 The interpolator
232 """
233 if varname in self._interpolators:
234 return self._interpolators[varname]
235 input_times = self.get('time')
236 if len(self.get(varname).shape) == 2:
237 if grid.kind in ('H', 'hybridH'):
238 input_coord = self.get('zh_forc', 0)
239 elif grid.kind in ('P', 'hybridP'):
240 input_coord = self.get('pa_forc', 0)
241 else:
242 raise Phyex1DError('Wrong grid kind')
243 interp = RegularGridInterpolator((input_times, input_coord),
244 self.get(varname),
245 bounds_error=False, fill_value=None)
246 else:
247 interp = RegularGridInterpolator((input_times, ),
248 self.get(varname),
249 bounds_error=False, fill_value=None)
250 self._interpolators[varname] = interp
251 return interp

399def _git(args, directory=None):
400 """Executes a git command (in directory), returns its standard output"""
401 cmd = ['git'] + ([] if directory is None else ['-C', directory]) + args

414def install_cases(directory=None):
415 """
416 Installs (clone) or updates (fetch and reset to the remote branch) the DEPHY-SCM
417 repository containing the cases in the common format (the repository is big, about
418 1 GB).
419
420 :param directory: directory of the repository (cases_directory() by default)
421 :return: directory of the repository
422 """
423 directory = cases_directory() if directory is None else directory
424 if os.path.islink(directory):
425 raise Phyex1DError(f"{directory} is a link to {os.readlink(directory)}: the "
426 "repository it points to must be updated by its owner (or the "
427 "link must be removed before downloading)")
428 if os.path.isdir(os.path.join(directory, '.git')):
429 # Update robust to a rewritten history (forced push): the local branch is reset
430 # to the remote one (local modifications of the repository are lost)
431 _git(['fetch', 'origin'], directory)
432 _git(['reset', '--hard', '@{upstream}'], directory)
433 else:
434 os.makedirs(os.path.dirname(os.path.abspath(directory)), exist_ok=True)
435 _git(['clone', DEPHY_SCM_URL, directory])
436 return directory

439def link_cases(path, directory=None):
440 """
441 Uses an existing DEPHY-SCM repository (or a copy of it): a symbolic link to it is
442 created instead of downloading the repository.
443
444 :param path: path to the existing DEPHY-SCM repository
445 :param directory: path of the link (cases_directory() by default); an existing link
446 is replaced, an existing directory (downloaded repository) is kept
447 and an error is raised
448 :return: path of the link
449 """
450 directory = cases_directory() if directory is None else directory
451 path = os.path.abspath(path)
452 if not os.path.isdir(path):
453 raise Phyex1DError(f"{path} is not a directory")
454 if not glob.glob(os.path.join(path, '*', '*', '*_SCM_driver.nc')):
455 raise Phyex1DError(f"{path} does not contain DEPHY-SCM cases "
456 "(<CASE>/<SUBCASE>/*_SCM_driver.nc files)")
457 if os.path.islink(directory):
458 os.remove(directory)
459 elif os.path.exists(directory):
460 raise Phyex1DError(f"{directory} already exists and is not a link (downloaded "
461 "repository?), remove it before creating the link")
462 os.makedirs(os.path.dirname(os.path.abspath(directory)), exist_ok=True)
463 os.symlink(path, directory)
464 return directory

467def list_cases(directory=None):
468 """
469 Lists the cases available in the DEPHY-SCM repository.
470
471 :param directory: directory of the repository (cases_directory() by default)
472 :return: list of tuples (case, subcase, path of the driver file)
473 """
474 directory = cases_directory() if directory is None else directory
475 if not os.path.isdir(directory):
476 raise Phyex1DError(f"The DEPHY-SCM repository is not installed in {directory}, "
477 "use phyex1d-cases_install")
478 result = []
479 for path in glob.glob(os.path.join(directory, '*', '*', '*_SCM_driver.nc')):
480 match = _DRIVER_RE.match(os.path.relpath(path, directory))
481 if match:
482 result.append((match.group('case'), match.group('subcase'), path))
483 return sorted(result)

486def main_cases_install():
487 """Entry point of the phyex1d-cases_install command"""
488 parser = argparse.ArgumentParser(
489 description=textwrap.fill(
490 'Installs the DEPHY-SCM cases (in the common format) in '
491 f'{cases_directory()} (the PHYEXCONF environment variable can be used to '
492 'change the root directory, ~/.phyex by default)'),
493 epilog=commands_help(), formatter_class=argparse.RawDescriptionHelpFormatter)
494 subparsers = parser.add_subparsers(dest='command', required=True)
495 subparsers.add_parser('download',
496 help='download (clone) or update the DEPHY-SCM repository (big: '
497 'about 1 GB)')
498 link = subparsers.add_parser('link', help='use an existing DEPHY-SCM repository (a '
499 'symbolic link to it is created)')
500 link.add_argument('path', help='path to the existing DEPHY-SCM repository')
501 args = parser.parse_args()
502 try:
503 if args.command == 'download':
504 directory = install_cases()
505 print(f'DEPHY-SCM cases installed in {directory}')
506 else:
507 directory = link_cases(args.path)
508 print(f'{directory} is a link to {os.path.abspath(args.path)}')
509 except Phyex1DError as err:
510 sys.exit(f'Error: {err}')
