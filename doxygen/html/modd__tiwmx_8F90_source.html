64! Fills the tables with the analytic formulas (internal functions below)
65 USE yomhook, ONLY : lhook, dr_hook, jphook

88 tiwmx%AF3TAB(jk) = af3(ztemp)
89 tiwmx%BB3TAB(jk) = bb3(ztemp)
90 tiwmx%BB3WTAB(jk) = bb3w(ztemp)
91 tiwmx%REDINTAB(jk) = redin(ztemp)
92
93 IF (ztemp.GT.273.15) THEN
94! Avoid ice calculation above freezing level:
95 tiwmx%ESTABI(jk) = esatw(ztemp)
96 tiwmx%DESTABI(jk) = desdtw(ztemp)

98! Functions for ice crystal growth:
99 tiwmx%A2TAB(jk) = aa2(ztemp)
100 tiwmx%A2WTAB(jk) = aa2w(ztemp)
101 tiwmx%AM3TAB(jk) = 0.
102 tiwmx%AF3TAB(jk) = 0.
103 tiwmx%BB3TAB(jk) = bb3(ztemp)
104 tiwmx%BB3WTAB(jk) = bb3w(ztemp)

112!
113! Analytic formulas, used only to fill the tables (the functions used during the simulation,
114! which read the tables, are in tiwmx.func.h)
115!
116!
117 ELEMENTAL REAL FUNCTION esatw(TT)
118 USE modd_cst, ONLY : xalpw, xbetaw, xgamw
119 REAL,INTENT(IN) :: tt
120 esatw = exp( xalpw - xbetaw/tt - xgamw*log(tt) )
121 END FUNCTION esatw
122!
123! pure saturation pressure over ice for tt <0 C,
124! esatw otherwise.
125!
126 ELEMENTAL REAL FUNCTION esati(TT)
127 USE modd_cst, ONLY : xalpw, xbetaw, xgamw, xalpi,xbetai,xgami, xtt
128 REAL,INTENT(IN) :: tt
129 esati = ( 0.5 + sign(0.5,tt-xtt) )*exp( xalpw - xbetaw/tt - xgamw*log(tt) ) - &
130 & ( sign(0.5,tt-xtt) - 0.5)*exp( xalpi - xbetai/tt - xgami*log(tt) )
131 END FUNCTION esati
132!
133! pure saturation pressure over water
134 ELEMENTAL REAL FUNCTION desdtw(TT)
135 USE modd_cst, ONLY : xbetaw, xgamw
136 REAL,INTENT(IN) :: tt
137 desdtw = esatw(tt)*(xbetaw/tt - xgamw)/tt
138 END FUNCTION desdtw
139
140 ELEMENTAL REAL FUNCTION desdti(TT)
141 USE modd_cst, ONLY : xbetai, xgami, xtt
142 REAL,INTENT(IN) :: tt
143 desdti = ( 0.5 + sign(0.5,tt-xtt) )*desdtw(tt) - &
144 & ( sign(0.5,tt-xtt) - 0.5)*esati(tt)*(xbetai/tt - xgami)/tt
145 END FUNCTION desdti
146
147! Ice crystal function:
148 ELEMENTAL REAL FUNCTION aa2(TT)
149 USE modd_cst, ONLY : xlstt, xcpv, xci, xtt, xrv
150 REAL,INTENT(IN) :: tt
151 aa2 = ( xlstt + (xcpv-xci)*(tt-xtt) )**2 / &
152 & (2.38e-2 + 0.0071e-2 *(tt - xtt))/(tt**2*xrv)
153 END FUNCTION aa2
154
155! Water droplet function:
156 ELEMENTAL REAL FUNCTION aa2w(TT)
157 USE modd_cst, ONLY : xlvtt, xcpv, xcl, xtt, xrv
158 REAL,INTENT(IN) :: tt
159 aa2w = ( (xlvtt+ (xcpv-xcl)*(tt-xtt))**2)/ &
160 & (2.38e-2 + 0.0071e-2 *(tt - xtt))/(tt**2*xrv)
161 END FUNCTION aa2w
162
163! Ice crystal function:
164 ELEMENTAL REAL FUNCTION bb3(TT)
165 USE modd_cst, ONLY : xtt, xrv, xp00
166 REAL,INTENT(IN) :: tt
167 bb3 = xrv/(0.211e-4 * (tt/xtt)**1.94 * xp00)*tt/esati(tt)
168 END FUNCTION bb3
169
170! Water droplet function:
171 ELEMENTAL REAL FUNCTION bb3w(TT)
172 USE modd_cst, ONLY : xtt, xrv, xp00
173 REAL,INTENT(IN) :: tt
174 bb3w = xrv/(0.211e-4 * (tt/xtt)**1.94 * xp00)*tt/esatw(tt)
175 END FUNCTION bb3w
176
177! Meyers IN concentration function:
178 ELEMENTAL REAL FUNCTION am3(TT)
179 REAL,INTENT(IN) :: tt
180 am3 = 1000.*exp(12.96*(esatw(tt)/esati(tt) -1.) -0.639)
181 END FUNCTION am3
182
183! Fletchers IN concentration function:
184 ELEMENTAL REAL FUNCTION af3(TT)
185 USE modd_cst, ONLY : xtt
186 REAL,INTENT(IN) :: tt
187 af3 = 0.01*exp(0.6*(xtt-tt))
188 END FUNCTION af3
189
190! Function for IN concentration reduction between 0 and -20 C:
191 ELEMENTAL REAL FUNCTION redin(TT)
192 USE modd_cst, ONLY : xtt
193 REAL,INTENT(IN) :: tt
194 REAL zzt
195 zzt = max(0., min(1.,(xtt - tt)/20.))
196 redin = 1.- (1.-zzt)/(zzt**3 + (1.-zzt)**3)**.333
197 END FUNCTION redin
