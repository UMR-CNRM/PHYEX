78 def add_vertical_coordinate(self, state):
79 """
80 Add vertical coordinate to the state
81
82 Parameters
83 ----------
84 state : dict
85 Current state
86 """
87 state['Z_mass'] = self.grid.get_altitude('MASS', state, self.prognostic_variables)
88 state['P'] = self.grid.get_pressure('MASS', state, self.prognostic_variables)
89 state['Z_flux'] = self.grid.get_altitude('FLUX', state, self.prognostic_variables)
90 top = -1 if self.grid.ascending else 0
91 if numpy.all(state['Z_flux'][top] == numpy.inf):
92 state['Z_flux'][top] = state['Z_flux'][-2 if self.grid.ascending else 1]

94 def add_conversions(self, state):
95 """
96 Add derived variables to the state
97
98 Parameters
99 ----------
100 state : dict
101 Current state
102 """
103 pressure_mass = self.grid.get_pressure('MASS', state, self.prognostic_variables)
104 exner = (pressure_mass / 1.E5) ** (self.cst.Rd / self.cst.Cpd)
105 if 'T' in self.prognostic_variables:
106 state['Theta'] = state['T'] / exner
107 elif 'Theta' in self.prognostic_variables:
108 state['T'] = state['Theta'] * exner
109
110 if 'qv' in self.prognostic_variables:
111 qt = state['qv'].copy()
112 for var in ('qc', 'qi', 'qr', 'qs', 'qg', 'qh'):
113 if var in state:
114 qt += state[var]
115 state['qt'] = qt
116 for v in ('v', 'c', 'i', 'r', 's', 'g', 'h'):
117 if 'q' + v in state:
118 state['r' + v] = state['q' + v] / (1. - qt)
119 elif 'rv' in self.prognostic_variables:
120 rt = state['rv'].copy()
121 for var in ('rc', 'ri', 'rr', 'rs', 'rg', 'rh'):
122 if var in state:
123 rt += state[var]
124 state['rt'] = rt
125 for v in ('v', 'c', 'i', 'r', 's', 'g', 'h'):
126 if 'r' + v in state:
127 state['q' + v] = state['r' + v] / (1. + rt)

129 def build_init_state(self, init_state):
130 """
131 Build the initial state
132
133 Parameters
134 ----------
135 init_state : dict
136 Initial state dictionary to populate
137
138 Returns
139 -------
140 dict
141 Populated initial state
142 """
143 init_state = super().build_init_state(init_state)
144
145 with self.case:
146 # Surface fields
147 for var in ('Zs', 'Ps', 'sshf', 'slhf', 'z0', 'z0h', 'z0q', 'Tskin', 'Ts', 'ustar',
148 'SolarIrradiance', 'albedo', 'emissivity', 'sza', 'lat', 'lon'):
149 if self.case.contains(var):
150 init_state[var] = self.case.get(var, 0)
151 else:
152 logging.warning('%s not found in the case description!', var)
153
154 # We perform the interpolation on the natural vertical coordinate
155 if self.grid.kind in ('H', 'hybridH'):
156 input_coord = self.case.get('zh', 0) + self.case.get('orog', 0)
157 output_coord = self.grid.get_altitude('MASS', init_state, self.prognostic_variables)
158 elif self.grid.kind in ('P', 'hybridP'):
159 input_coord = self.case.get('pa', 0)
160 output_coord = self.grid.get_pressure('MASS', init_state, self.prognostic_variables)
161 else:
162 raise Phyex1DError('Wrong grid kind')
163
164 # Interpolation of initial profiles
165 for var in self.prognostic_variables:
166 if var in ('qc', 'qi', 'qr', 'qs', 'qg', 'qh',
167 'rc', 'ri', 'rr', 'rs', 'rg', 'rh',
168 'w') and not self.case.contains(var):
169 init_state[var] = numpy.zeros(output_coord.shape)
170 logging.warning('%s not found in the case description!', var)
171 else:
172 interp = RegularGridInterpolator((input_coord, ), self.case.get(var, 0),
173 bounds_error=False, fill_value=None)
174 init_state[var] = interp(output_coord)
175 init_state['tke'] = numpy.maximum(self.tkemin, init_state['tke'])
176
177 # Vertical coordinates
178 self.add_vertical_coordinate(init_state)
179
180 # Conversions
181 self.add_conversions(init_state)
182
183 # Arrays needed to keep memory from one timestep to the other
184 for var in ('ni', 'sigs', 'WEIGHT_MF_CLOUD', 'CF_MF', 'rc_MF', 'ri_MF',
185 'HLC_HRC_MF', 'HLC_HCF_MF', 'HLI_HRI_MF', 'HLI_HCF_MF'):
186 init_state[var] = numpy.zeros(init_state['P'].shape)
187
188 # Initial value
189 for var in ('CF', 'dt_rad_lw', 'dt_rad_sw'):
190 init_state[var] = numpy.zeros(init_state['P'].shape)
191 for var in ('lw_up', 'lw_dn', 'sw_up', 'sw_dn'):
192 init_state[var] = numpy.zeros((init_state['P'].shape[0] + 1, ))
193 for var in ('sfrv', 'swf', 'sfth', 'sshf', 'slhf'):
194 if var not in init_state:
195 init_state[var] = numpy.zeros((1, ))
196
197 return init_state

199 def forcing(self, state, timestep, timestep_number):
200 """
201 Apply large-scale forcings to the state
202
203 Parameters
204 ----------
205 state : dict
206 Dictionary holding all the variables.
207 timestep : float
208 Timestep (s)
209 timestep_number : int
210 Current time step number
211
212 Returns
213 -------
214 dict
215 Updated state after forcing
216 """
217 time = timestep * timestep_number
218 # All forcing variable are taken on mass levels
219 if self.grid.kind in ('H', 'hybridH'):
220 output_coord = self.grid.get_altitude('MASS', state, self.prognostic_variables)
221 elif self.grid.kind in ('P', 'hybridP'):
222 output_coord = self.grid.get_pressure('MASS', state, self.prognostic_variables)
223 else:
224 raise Phyex1DError('Wrong grid kind')
225
226 with self.case:
227
228 # Temperature advection and nudging, radiative tendency
229 if 'Theta' in self.prognostic_variables:
230 if self.case.adv_theta == 1:
231 interp = self.case.get_interpolator('tntheta_adv', self.grid)
232 state['Theta'] += interp(numpy.column_stack(([time] * len(output_coord),
233 output_coord))) * timestep
234 if self.case.nudging_theta == 1:
235 raise NotImplementedError('Nudging theta')
236 if self.case.radiation == 'tend':
237 interp = self.case.get_interpolator('tntheta_rad', self.grid)
238 state['Theta'] += interp(numpy.column_stack(([time] * len(output_coord),
239 output_coord))) * timestep
240 elif 'T' in self.prognostic_variables:
241 if self.case.adv_ta == 1:
242 interp = self.case.get_interpolator('tnta_adv', self.grid)
243 state['T'] += interp(numpy.column_stack(([time] * len(output_coord),
244 output_coord))) * timestep
245 if self.case.nudging_ta == 1:
246 raise NotImplementedError('Nudging ta')
247 if self.case.radiation == 'tend':
248 interp = self.case.get_interpolator('tnta_rad', self.grid)
249 state['T'] += interp(numpy.column_stack(([time] * len(output_coord),
250 output_coord))) * timestep
251
252 # Hydrometeors advection and nudging
253 if 'rv' in self.prognostic_variables:
254 if self.case.adv_rv == 1:
255 interp = self.case.get_interpolator('tnrv_adv', self.grid)
256 state['rv'] += interp(numpy.column_stack(([time] * len(output_coord),
257 output_coord))) * timestep
258 if self.case.nudging_rv == 1:
259 raise NotImplementedError('Nudging rv')
260 elif 'qv' in self.prognostic_variables:
261 if self.case.adv_qv == 1:
262 interp = self.case.get_interpolator('tnqv_adv', self.grid)
263 state['qv'] += interp(numpy.column_stack(([time] * len(output_coord),
264 output_coord))) * timestep
265 if self.case.nudging_qv == 1:
266 raise NotImplementedError('Nudging qv')
267
268 # Wind advection and nudging
269 if self.case.adv_ua == 1:
270 interp = self.case.get_interpolator('tnua_adv', self.grid)
271 state['u'] += interp(numpy.column_stack(([time] * len(output_coord),
272 output_coord))) * timestep
273 if self.case.adv_va == 1:
274 interp = self.case.get_interpolator('tnva_adv', self.grid)
275 state['v'] += interp(numpy.column_stack(([time] * len(output_coord),
276 output_coord))) * timestep
277 if self.case.nudging_ua == 1:
278 raise NotImplementedError('Nudging u')
279 if self.case.nudging_va == 1:
280 raise NotImplementedError('Nudging v')
281 if self.case.forc_wa == 1:
282 interp = self.case.get_interpolator('wa', self.grid)
283 w = interp(numpy.column_stack(([time] * len(output_coord),
284 output_coord)))
285 z_mass = self.grid.get_altitude('MASS', state, self.prognostic_variables)
286 for var in [var for var in self.prognostic_variables if var != 'w']:
287 grad = numpy.gradient(state[var], z_mass)
288 state[var] += - grad * w * timestep
289 if var in ('qv', 'qc', 'qr', 'qi', 'qs', 'qg', 'qh',
290 'rv', 'rc', 'rr', 'ri', 'rs', 'rg', 'rh',
291 'tke'):
292 state[var] = numpy.maximum(state[var], 0.)
293 if self.case.forc_geo == 1:
294 interp_ug = self.case.get_interpolator('ug', self.grid)
295 interp_vg = self.case.get_interpolator('vg', self.grid)
296 f = 2. * self.cst.omega * numpy.sin(numpy.radians(state['lat']))
297 state['u'] += +f * (state['v'] - interp_vg(
298 numpy.column_stack(([time] * len(output_coord), output_coord)))) * timestep
299 state['v'] += -f * (state['u'] - interp_ug(
300 numpy.column_stack(([time] * len(output_coord), output_coord)))) * timestep
301
302 # Surface
303 for surf_var in ('Ts', 'Tskin', 'Ps'):
304 if surf_var in state:
305 interp = self.case.get_interpolator(
306 {'Ts': 'ts_forc', 'Tskin': 'tskin', 'Ps': 'ps_forc'}[surf_var], self.grid)
307 state[surf_var] = interp([time])[0]
308 if self.case.surface_forcing_temp == 'surface_flux':
309 interp = self.case.get_interpolator('hfss', self.grid)
310 state['sshf'] = interp([time])[0]
311 if self.case.surface_forcing_moisture == 'surface_flux':
312 interp = self.case.get_interpolator('hfls', self.grid)
313 state['slhf'] = interp([time])[0]
314 for var in ('z0', 'z0h', 'z0q'):
315 if var in state:
316 interp = self.case.get_interpolator(var, self.grid)
317 state[var] = interp([time])[0]
318
319 # Other fields
320 interp = self.case.get_interpolator('lat', self.grid)
321 state['lat'] = interp([time])[0]
322 interp = self.case.get_interpolator('lon', self.grid)
323 state['lon'] = interp([time])[0]
324
325 return state
