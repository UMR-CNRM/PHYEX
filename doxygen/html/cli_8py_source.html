21def convert_option(cls_name, key, value, default):
22 """
23 Conversion of a command line value, according to the type of the default value
24 of the option in the physics class
25 :param cls_name: name of the physics class (for error messages)
26 :param key: option name
27 :param value: string to convert
28 :param default: default value of the option
29 :return: converted value
30 """
31 if isinstance(default, bool):
32 if value not in ('True', 'False'):
33 raise Phyex1DError(f"Option {key} of {cls_name} must be True or False")
34 return value == 'True'
35 if isinstance(default, (int, float)):
36 try:
37 return type(default)(value)
38 except ValueError as err:
39 raise Phyex1DError(f"Option {key} of {cls_name} must be a "
40 f"{type(default).__name__}") from err
41 return value

59def physics_options_help():
60 """
61 Description of the options of the physics classes (OPTIONS_HELP class attribute) with
62 their default values; classes having the same options are grouped
63 """
64 groups = {}
65 for cls_name, cls in PHYSICS_CLASSES.items():
66 options = tuple((key, repr(default), cls.OPTIONS_HELP.get(key, ''))
67 for key, default in physics_options(cls).items())
68 groups.setdefault(options, []).append(cls_name)
69 lines = []
70 for options, cls_names in groups.items():
71 lines.append(f"- for {', '.join(cls_names)}:")
72 for key, default, description in options:
73 lines.append(textwrap.fill(f' - {key} (default: {default}): {description}',
74 width=90, subsequent_indent=' ',
75 break_on_hyphens=False))
76 return '\n'.join(lines)

132def input_file(case):
133 """
134 Path of the netCDF file describing the case
135 :param case: path to a netCDF file or name of an installed DEPHY-SCM case
136 (CASE/SUBCASE)
137 :return: path to the netCDF file
138 """
139 if os.path.isfile(case):
140 return case
141 for case_name, subcase_name, path in list_cases():
142 if case.strip('/') == case_name + '/' + subcase_name:
143 return path
144 raise Phyex1DError(f"{case} is neither a file nor an installed DEPHY-SCM case "
145 "(see phyex1d-cases_list)")

167def experiments_from_args(args):
168 """List of experiment descriptions (for ExperimentRunner) from the --exp options"""
169 experiments = []
170 for expe in args.experiments or [[]]:
171 values = dict(arg.split('=', 1) for arg in expe if not arg.startswith('attr='))
172 cls_name = values.get('class', DEFAULT_CLASS)
173 if cls_name not in PHYSICS_CLASSES:
174 raise Phyex1DError(f"Unknown physics class {cls_name}, allowed classes are: " +
175 ', '.join(PHYSICS_CLASSES))
176 options = physics_options(PHYSICS_CLASSES[cls_name])
177 expe_descr = {}
178 attrs = {}
179 for arg in expe:
180 key, val = arg.split('=', 1)
181 if key == 'attr':
182 attr, attrval = val.split(':', 1)
183 if attr in ('radiation', ):
184 attrs[attr] = str(attrval)
185 else:
186 raise Phyex1DError(f"This experiment attribute is not implemented: {attr}")
187 elif key == 'dt':
188 expe_descr[key] = float(val)
189 elif key in ('name', 'grid', 'class'):
190 expe_descr[key] = val
191 elif key in options:
192 expe_descr[key] = convert_option(cls_name, key, val, options[key])
193 else:
194 raise Phyex1DError(f"Option {key} does not exist for {cls_name}, allowed "
195 "options are: dt, name, grid, class, attr, " +
196 ', '.join(options))
197 expe_descr['attrs'] = attrs
198 experiments.append(expe_descr)
199 return experiments

208def plot(ph1d, plot_args):
209 """
210 Plots the evolution of variables
211 :param ph1d: ExperimentRunner instance
212 :param plot_args: arguments of a --plot option (file name, options and variables)
213 """
214 fig, ax = plt.subplots()
215 filename = plot_args[0]
216 var_names = []
217 options = {}
218 ylim = [None, None] # min max for y-axis
219 vlim = [None, None] # min and max value
220 for arg in plot_args[1:]:
221 if '=' in arg:
222 key, value = arg.split('=', 1)
223 if key in ('y_var', 'only_param_tag', 'title', 'xlabel', 'ylabel',
224 'legend'):
225 # Strings
226 options[key] = value
227 elif key in ('mfactor', 'linewidth'):
228 # Floats
229 options[key] = float(value)
230 elif key in ('enable_contourf', 'switch_cls', 'colorbar', 'clabel'):
231 # Booleans
232 options[key] = value == 'True'
233 elif key in ('ymin', 'ymax'):
234 if key == 'ymin':
235 ylim[0] = float(value)
236 else:
237 ylim[1] = float(value)
238 elif key in ('vmin', 'vmax'):
239 if key == 'vmin':
240 vlim[0] = float(value)
241 else:
242 vlim[1] = float(value)
243 else:
244 raise Phyex1DError(f"This plot option does not exist: {key}")
245 else:
246 var_names.append(arg)
247 kwargs_contourf = {}
248 if vlim[0] is not None:
249 kwargs_contourf['vmin'] = vlim[0]
250 if vlim[1] is not None:
251 kwargs_contourf['vmax'] = vlim[1]
252 if len(kwargs_contourf) != 0:
253 options['kwargs_contourf'] = kwargs_contourf
254 ph1d.plot_evol(ax, var_names, **options)
255 if options.get('y_var', 'X') == 'P':
256 ax.invert_yaxis()
257 ax.set_ylim(ylim)
258 fig.savefig(filename)
259 plt.close(fig)
