1#!/usr/bin/env python3
2"""
3Transformation of a PHYEX source file by pyfortool, executed at build time
4(see the top-level CMakeLists.txt, PHYEX_TRANSFORMATION option):
5 phyex_transform.py --transformation <name> [--pyfortool <exe>] <input> <output>
6
7The new source code is generated by pyfortool with arguments depending on the
8transformation, complemented by '--optsByEnv PYFT_OPTS' if the PYFT_OPTS
9environment variable is not empty. It is written in a temporary file which is
10renamed into <output> when the transformation succeeds.
11
12The list of available transformations is printed with --list.
13"""
14
15import argparse
16import os
17import subprocess
18import sys
19
20# Transformation name -> pyfortool arguments
21TRANSFORMATIONS = {
22 'IAL': ['--wrapH', '--removeACC', '--shugradTransfo', '--mnhExpand', '--splitLongLines'],
23 # IAL without the transformation of shuman/gradient functions into subroutine calls
24 'IAL_NOSHUGRAD': ['--wrapH', '--removeACC', '--mnhExpand', '--splitLongLines'],
25 # IAL without the expansion of the mnh_expand directives (code kept in array syntax)
26 'IAL_NOEXPAND': ['--wrapH', '--removeACC', '--shugradTransfo', '--splitLongLines'],
27}
28
29# Name of the environment variable containing additional pyfortool arguments
30OPTS_ENV = 'PYFT_OPTS'
31
32
33def main():

39 help='Transformation to apply')

42 help='Unset PYTHONPATH when running pyfortool')

46
47 if args.list:

49 return 0

61 if os.environ.get(OPTS_ENV, ''):
62 cmd += ['--optsByEnv', OPTS_ENV]
63 cmd += [args.input, tmp]
64 env = None
65 if args.unset_pythonpath:
66 env = dict(os.environ)
67 env.pop('PYTHONPATH', None)
68 try:
69 status = subprocess.run(cmd, env=env, check=False).returncode
70 except OSError as err:
71 print(f'phyex_transform: unable to execute {cmd[0]}: {err}', file=sys.stderr)
72 status = 1
73 if status != 0 or not os.path.exists(tmp):
74 print(f'phyex_transform: transformation {args.transformation} of {args.input} failed: ' +
75 ' '.join(cmd), file=sys.stderr)
76 if os.path.exists(tmp):
77 os.remove(tmp)
