56def simplify_code(src):
57 # Remove comments, remove double spaces, join lines using continuation marker (&)
58 lines = []
59 buf = ""
60 for line in src:
61 line = line.strip()
62 if line.startswith("!"):
63 continue
64 while " " in line:
65 line = line.replace(" ", " ")
66 if not reg_case_is_important.match(line):
67 line = line.upper()
68 if "!" in line and not line.startswith("#"):
69 # When '!' is in the middle of a line, most of the time what follow it is comment.
70 # But sometimes, it is a negation in an ifdef, see: #if ! defined foo
71 line = line.split("!")[0].strip()
72 if line.endswith("&"):
73 buf += line.replace("&", "")
74 else:
75 lines.append(buf + line.replace("&", "") + "\n")
76 buf = ""
77 return lines

106def remove_unsed_modules(lines):
107 use_only = {} # line_idx -> "use only" values
108 # list of lines with used module without an only clause,
109 # or imported value that are not used in the procedure signature
110 to_rm = []
111
112 # Retrieve all explicitly imported values (eg. use foo, only: stuff)
113 # If there is no 'only' clause, the line is mark for deletion
114 for idx, line in enumerate(lines):
115 if line.startswith("USE "):
116 matches = reg_use_module_only.match(line)
117 if not matches:
118 to_rm.append(idx)
119 else:
120 use_only[idx] = [v.strip() for v in matches.group(1).split(",") if v.strip()]
121
122 # Identifiers used in the variable declarations (whole words only, string
123 # constants and derived type components removed: 'TM06' does not use TM06,
124 # LRMC01 does not use RMC01)
125 used = set()
126 for line in lines:
127 if reg_is_variable_decl.match(line):
128 line = reg_component.sub("", reg_string.sub("", line))
129 used.update(reg_identifier.findall(line))
130
131 # Only the imported values used in a variable declaration are kept, a line
132 # without any of them is removed
133 for idx, values in use_only.items():
134 kept = []
135 for val in values:
136 # renaming (local => used): the local name is the one used
137 local = val.split("=>")[0].strip()
138 # generic operators and assignments are always kept
139 if local.startswith(("OPERATOR", "ASSIGNMENT")) or local in used:
140 kept.append(val)
141 if kept:
142 module = reg_use_module_name.match(lines[idx]).group(1)
143 lines[idx] = f"USE {module}, ONLY: {', '.join(kept)}\n"
144 else:
145 to_rm.append(idx)
146
147 for idx in reversed(sorted(to_rm)):
148 lines.pop(idx)

151def get_proc_signature(line):
152 # Retrieve type of procedure, name, arguments (with result if any)
153 matches = reg_proc_name.match(line)
154 proc_type = matches.group(1)
155 proc_name = matches.group(2)

158 matches = reg_proc_args.search(line)

170def break_long_lines(lines):
171 newlines = []
172 for line in lines:
173 while len(line) > 80:
174 break_idx_space = line[0:80].rfind(" ")
175 break_idx_comma = line[0:80].rfind(",")
176
177 # Find which one let us have the longest line of less than 80 chars
178 if break_idx_comma > break_idx_space:
179 break_idx = break_idx_comma
180 else:
181 break_idx = break_idx_space
182
183 # If no space or comma was found, force break the line at 80 chars
184 if break_idx == -1:
185 break_idx = 80
186
187 newlines.append(line[:break_idx] + "&\n")
188 line = line[break_idx:]
189 newlines.append(line)
190 return newlines
