7# Start of a procedure, with its optional prefixes (PURE, ELEMENTAL, RECURSIVE...
8# and, for a function, its type: REAL, REAL(KIND=JPRB), TYPE(T), CHARACTER*8...)
9reg_type_spec = (r"(?:INTEGER|REAL|DOUBLE\s*PRECISION|COMPLEX|CHARACTER|LOGICAL|TYPE|CLASS)"
10 r"\s*(?:\‍((?:[^()]|\‍([^()]*\‍))*\‍)|\*\s*\w+)?")
11reg_proc_prefix = (r"(?:(?:IMPURE|PURE|ELEMENTAL|RECURSIVE|NON_RECURSIVE)\s+|"
12 + reg_type_spec + r"\s*)*")
13reg_proc_name = re.compile(r"^\s*" + reg_proc_prefix + r"(SUBROUTINE|FUNCTION)\s+(\w+)",
14 re.IGNORECASE)
15# End of a procedure (a bare END can only end a program unit)
16reg_proc_end = re.compile(r"^\s*END\s*((SUBROUTINE|FUNCTION)\b|$)", re.IGNORECASE)
17reg_proc_args = re.compile(r"\‍(([\w\s,]+)\‍)", re.IGNORECASE)
18reg_func_result = re.compile(r"RESULT\s*\‍(([\w\s]+)\‍)", re.IGNORECASE)
19reg_use_module_only = re.compile(r"(?:^\s*use\s+\w+\s*,\s*only\s*:)(.*)", re.IGNORECASE)
20reg_use_module_name = re.compile(r"^\s*use\s+(\w+)", re.IGNORECASE)
21reg_variable_name_simple = re.compile(r"::\s*(\w+)", re.IGNORECASE)
22reg_string = re.compile(r"'[^']*'|\"[^\"]*\"")
23reg_component = re.compile(r"%\s*\w+") # derived type components (D%NIJT)
24reg_identifier = re.compile(r"[A-Z_][A-Z0-9_]*")
25
26# Macro directives and Fortran include statements are case-sensitive so they must never be uppercased
27reg_case_is_important = re.compile(r"\s*#|\s*INCLUDE", re.IGNORECASE)

68def simplify_code(src):
69 # Remove comments, remove double spaces, join lines using continuation marker (&)
70 lines = []
71 buf = ""
72 for line in src:
73 line = line.strip()
74 if line.startswith("!"):
75 continue
76 while " " in line:
77 line = line.replace(" ", " ")
78 if not reg_case_is_important.match(line):
79 line = line.upper()
80 if "!" in line and not line.startswith("#"):
81 # When '!' is in the middle of a line, most of the time what follow it is comment.
82 # But sometimes, it is a negation in an ifdef, see: #if ! defined foo
83 line = line.split("!")[0].strip()
84 if line.endswith("&"):
85 buf += line.replace("&", "")
86 else:
87 lines.append(buf + line.replace("&", "") + "\n")
88 buf = ""
89 return lines

118def remove_unsed_modules(lines):
119 use_only = {} # line_idx -> "use only" values
120 # list of lines with used module without an only clause,
121 # or imported value that are not used in the procedure signature
122 to_rm = []
123
124 # Retrieve all explicitly imported values (eg. use foo, only: stuff)
125 # If there is no 'only' clause, the line is mark for deletion
126 for idx, line in enumerate(lines):
127 if line.startswith("USE "):
128 matches = reg_use_module_only.match(line)
129 if not matches:
130 to_rm.append(idx)
131 else:
132 use_only[idx] = [v.strip() for v in matches.group(1).split(",") if v.strip()]
133
134 # Identifiers used in the variable declarations (whole words only, string
135 # constants and derived type components removed: 'TM06' does not use TM06,
136 # LRMC01 does not use RMC01)
137 used = set()
138 for line in lines:
139 if reg_is_variable_decl.match(line):
140 line = reg_component.sub("", reg_string.sub("", line))
141 used.update(reg_identifier.findall(line))
142
143 # Only the imported values used in a variable declaration are kept, a line
144 # without any of them is removed
145 for idx, values in use_only.items():
146 kept = []
147 for val in values:
148 # renaming (local => used): the local name is the one used
149 local = val.split("=>")[0].strip()
150 # generic operators and assignments are always kept
151 if local.startswith(("OPERATOR", "ASSIGNMENT")) or local in used:
152 kept.append(val)
153 if kept:
154 module = reg_use_module_name.match(lines[idx]).group(1)
155 lines[idx] = f"USE {module}, ONLY: {', '.join(kept)}\n"
156 else:
157 to_rm.append(idx)
158
159 for idx in reversed(sorted(to_rm)):
160 lines.pop(idx)

163def get_proc_signature(line):
164 # Retrieve type of procedure, name, arguments (with result if any)
165 matches = reg_proc_name.match(line)
166 proc_type = matches.group(1)
167 proc_name = matches.group(2)
168
169 # arguments are searched after the procedure name (not in the type)
170 end_name = matches.end()
171 proc_args = []
172 matches = reg_proc_args.search(line, end_name)
173 if matches:
174 proc_args = [arg.strip() for arg in matches.group(1).split(",")]
175
176 if proc_type == "FUNCTION":
177 matches = reg_func_result.search(line, end_name)
178 if matches:
179 proc_args.append(matches.group(1).strip())
180
181 return proc_type, proc_name, proc_args

184def break_long_lines(lines):
185 newlines = []
186 for line in lines:
187 while len(line) > 80:
188 break_idx_space = line[0:80].rfind(" ")
189 break_idx_comma = line[0:80].rfind(",")
190
191 # Find which one let us have the longest line of less than 80 chars
192 if break_idx_comma > break_idx_space:
193 break_idx = break_idx_comma
194 else:
195 break_idx = break_idx_space
196
197 # If no space or comma was found, force break the line at 80 chars
198 if break_idx == -1:
199 break_idx = 80
200
201 newlines.append(line[:break_idx] + "&\n")
202 line = line[break_idx:]
203 newlines.append(line)
204 return newlines

207def check_procedures(f90, lines):
208 # The file must contain exactly one procedure (outside a module), whose
209 # name is the file name: the interface module of xxx.F90 is MODI_XXX in
210 # modi_xxx.F90
211 names = [reg_proc_name.match(line).group(2) for line in lines
212 if reg_proc_name.match(line)]
213 if not names:
214 raise GenModiError(f"{f90}: no SUBROUTINE or FUNCTION found")
215 if len(names) > 1:
216 raise GenModiError(f"{f90}: several procedures found outside a module "
217 f"({', '.join(names)}), only one is allowed")
218 if names[0].upper() != f90.stem.upper():
219 raise GenModiError(f"{f90}: the procedure name ({names[0]}) must be the "
220 f"file name ({f90.stem})")

225 # No outdated file is left in case of error

228 lines = simplify_code(src)
229 if not is_modi_needed(lines):
230 modi_path.touch()

233 remove_contained_procedures(lines)
234 check_procedures(f90, lines)

236 modi_lines = []
237 proc_type = ""
238 proc_name = ""
239 proc_args = []
240 for line in lines:
241 if not proc_name:
242 if reg_proc_name.match(line):
243 proc_type, proc_name, proc_args = get_proc_signature(line)
244 modi_lines.append(f"MODULE MODI_{proc_name}\nINTERFACE\n")
245
246 matches = reg_is_still_decl.match(line)
247 if not matches and proc_name:

249 modi_lines.append(f"END INTERFACE\n")
250 modi_lines.append(f"END MODULE MODI_{proc_name}\n")
251 proc_name = ""
252 continue
253
254 if proc_name and reg_is_variable_decl.match(line) and "INTENT" not in line:
255 # if there are no intent, it might be a local variable
256 # but sometimes the intent is missing, so we check also
257 # if it's not in the list of arguments, it can also be the result of a function
258 matches = reg_variable_name_simple.search(line)
259 if matches:
260 var_name = matches.group(1)
261 if var_name not in proc_args and var_name != proc_name:
262 continue
263
264 if proc_name:
265 modi_lines.append(line)

268 remove_unsed_modules(modi_lines)
